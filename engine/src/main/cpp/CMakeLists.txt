cmake_minimum_required(VERSION 3.22.1)
project(xiangqi_native LANGUAGES CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS native_compile=3)
set(CMAKE_JOB_POOL_COMPILE native_compile)
get_filename_component(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/pikafish/src" ABSOLUTE)
include(upstream-sources.cmake)
find_package(Threads REQUIRED)
function(configure_engine target)
    target_include_directories(${target} PRIVATE "${UPSTREAM}" "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(${target} PRIVATE IS_64BIT NDEBUG)
    target_compile_options(${target} PRIVATE -O3)
    target_link_libraries(${target} PRIVATE Threads::Threads)
endfunction()
if(ANDROID)
    add_library(cpu_dispatch SHARED cpu_dispatch.cpp)
    target_link_options(cpu_dispatch PRIVATE "-Wl,-z,max-page-size=16384")
    foreach(variant base dotprod)
        add_library(pikafish_${variant} SHARED ${UPSTREAM_SOURCES} engine_bridge.cpp jni_bridge.cpp)
        configure_engine(pikafish_${variant})
        target_compile_definitions(pikafish_${variant} PRIVATE USE_NEON=8)
        target_link_options(pikafish_${variant} PRIVATE "-Wl,-z,max-page-size=16384")
        if(variant STREQUAL "dotprod")
            target_compile_definitions(pikafish_${variant} PRIVATE USE_NEON_DOTPROD)
            target_compile_options(pikafish_${variant} PRIVATE -march=armv8.2-a+dotprod)
        else()
            target_compile_options(pikafish_${variant} PRIVATE -march=armv8-a)
        endif()
    endforeach()
else()
    add_executable(host_probe ${UPSTREAM_SOURCES} engine_bridge.cpp host_probe.cpp)
    configure_engine(host_probe)
endif()
